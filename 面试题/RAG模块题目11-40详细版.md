## 参考答案

### 一、原理阐述

**核心概念**:
L4业务服务层是KQuery Pro七层架构中的核心编排层,负责协调底层存储、检索能力,对外提供统一的业务接口。模块化设计将复杂业务逻辑拆分为独立的功能模块,提高系统的可维护性和可扩展性。

**设计原则**:
1. **单一职责**:每个模块专注于特定功能域
2. **依赖倒置**:高层模块不依赖低层模块,两者都依赖抽象
3. **开闭原则**:对扩展开放,对修改关闭
4. **接口隔离**:客户端不应依赖它不需要的接口

**核心模块划分**:

```
app/services/
├── rag_service.py           # RAG统一服务门面
├── retriever.py              # 检索引擎编排
├── hyde_generator.py         # HyDE生成器
├── embedding.py              # 嵌入服务
├── chunking.py               # 文本切分
├── bm25_retriever.py         # BM25检索器
├── document_classifier.py    # 文档分类器
└── storage_service.py        # 存储服务
```

**模块间协作关系**:
```python

class PerformanceTracker:
    """模块性能追踪器"""

    def __init__(self, module_name: str):
        self.module_name = module_name
        self.metrics = {
            "call_count": 0,
            "total_time_ms": 0,
            "avg_time_ms": 0
        }

    def track(self, func):
        """装饰器模式追踪函数执行"""
        async def wrapper(*args, **kwargs):
            start_time = time.time()
            try:
                result = await func(*args, **kwargs)
                return result
            finally:
                elapsed = (time.time() - start_time) * 1000
                self.metrics["call_count"] += 1
                self.metrics["total_time_ms"] += elapsed
                self.metrics["avg_time_ms"] = (
                    self.metrics["total_time_ms"] / self.metrics["call_count"]
                )
        return wrapper
```

#### 题目12: GraphRAG引擎的异步编排实现 ⭐⭐

## 参考答案

### 一、原理阐述

**异步编排核心机制**:

GraphRAG引擎使用asyncio实现异步编排,将五个阶段的处理流程设计为异步流水线,提高并发性能和资源利用率。

**关键技术点**:
1. **async/await语法**: 协程实现非阻塞IO
2. **asyncio.gather**: 并行执行多个异步任务
3. **asyncio.create_task**: 创建后台任务
4. **asyncio.wait**: 等待任务完成并控制超时

**五阶段异步流程**:

```python
async def hybrid_search(self, query: str, top_k: int = 10):
    """异步编排的五阶段流程"""

    # 阶段1: 问题分类(快速同步执行)
    question_type = classify_question(query)

    # 阶段2: 实体抽取(异步IO)
    entities = await self.extractor.extract(query)

    # 阶段3: 图谱扩展 + 向量检索(并行执行)
    graph_task = asyncio.create_task(
        self.expander.expand(entities)
    )
    vector_task = asyncio.create_task(
        self.rag_service.search(query, top_k * 2)
    )

    # 并行等待两个任务
    graph_context, vector_results = await asyncio.gather(
        graph_task, vector_task
    )

    # 阶段4: 结果融合
    fused_results = await self.fusion_engine.fuse(
        vector_results, graph_context
    )

    # 阶段5: 返回结果
    return fused_results[:top_k]
```

```python
class ConcurrencyController:
    """并发控制器"""

    def __init__(self, max_concurrent: int = 10):
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.active_tasks = 0
        self.total_completed = 0

    async def execute_with_limit(self, coro):
        """在并发限制下执行协程"""
        async with self.semaphore:
            self.active_tasks += 1
            try:
                result = await coro
                return result
            finally:
                self.active_tasks -= 1
                self.total_completed += 1

# 在GraphRAG引擎中使用
class GraphRAGEngine:
    def __init__(self):
        self.concurrency_ctrl = ConcurrencyController(max_concurrent=20)

    async def batch_search(self, queries: List[str]):
        """批量检索,控制并发"""
        tasks = [
            self.concurrency_ctrl.execute_with_limit(
                self.hybrid_search(query)
            )
            for query in queries
        ]
        return await asyncio.gather(*tasks)
```

```python
class AsyncRetryPolicy:
    """异步重试策略"""

    def __init__(self, max_retries: int = 3, timeout: float = 5.0):
        self.max_retries = max_retries
        self.timeout = timeout

    async def execute_with_retry(self, coro, operation_name: str):
        """带重试和超时的执行"""
        for attempt in range(self.max_retries):
            try:
                # 使用wait_for添加超时
                result = await asyncio.wait_for(
                    coro(),
                    timeout=self.timeout
                )
                return result
            except asyncio.TimeoutError:
                logger.warning(
                    f"{operation_name} timeout on attempt {attempt + 1}"
                )
                if attempt == self.max_retries - 1:
                    raise
            except Exception as e:
                logger.error(
                    f"{operation_name} failed on attempt {attempt + 1}: {e}"
                )
                if attempt == self.max_retries - 1:
                    raise
                await asyncio.sleep(2 ** attempt)  # 指数退避

        raise RuntimeError(f"{operation_name} failed after retries")

# 应用到实体抽取
async def extract_entities_safe(self, query: str):
    """安全的实体抽取"""
    policy = AsyncRetryPolicy(max_retries=3, timeout=10.0)
    return await policy.execute_with_retry(
        lambda: self.extractor.extract(query),
        "EntityExtraction"
    )
```

```python
# 排查工具 - 异步依赖检测器
class AsyncDependencyDetector:
    """检测异步依赖和潜在死锁"""

    def __init__(self):
        self.task_deps = {}

    def track_dependency(self, task_name: str, depends_on: str):
        """记录任务依赖"""
        if task_name not in self.task_deps:
            self.task_deps[task_name] = set()
        self.task_deps[task_name].add(depends_on)

        # 检测循环依赖
        if self._has_cycle(task_name):
            raise RuntimeError(
                f"Detected async dependency cycle: {task_name} -> {depends_on}"
            )

    def _has_cycle(self, start: str) -> bool:
        """DFS检测依赖环"""
        visited = set()
        path = set()

        def dfs(node: str) -> bool:
            if node in path:
                return True  # 找到环
            if node in visited:
                return False

            visited.add(node)
            path.add(node)

            for dep in self.task_deps.get(node, []):
                if dfs(dep):
                    return True

            path.remove(node)
            return False

        return dfs(start)

# 使用示例
detector = AsyncDependencyDetector()

async def search_with_tracking(query: str):
    detector.track_dependency("search", "embedding")
    await embedding_service.embed(query)

    detector.track_dependency("search", "graph_expansion")
    await graph_expander.expand(query)
```

#### 题目13: 多路召回的并行执行策略 ⭐⭐

## 参考答案

### 一、原理阐述

**多路召回架构**:

KQuery Pro采用多路召回策略,同时使用向量检索、BM25关键词检索和图谱检索三种召回方式,提高检索召回率和多样性。

**并行执行优势**:
1. **降低延迟**: 多个召回路径并行执行,总耗时≈max(单路耗时)
2. **提高容错**: 单路召回失败不影响整体结果
3. **资源优化**: 充分利用多核CPU和IO并发

**并行策略选择**:

| 策略 | 实现方式 | 适用场景 | 优缺点 |
|-----|---------|---------|--------|
| asyncio.gather | 协程并行 | IO密集型 | ✅轻量 ✅低延迟 ❌受Python GIL限制 |
| ThreadPoolExecutor | 线程池 | 混合型 | ✅突破GIL ❌线程开销 |
| ProcessPoolExecutor | 进程池 | CPU密集型 | ✅真并行 ❌进程开销大 |
| asyncio.Queue | 生产消费 | 流式处理 | ✅解耦 ✅可控流量 |

### 二、场景应用

**案例1:三路召回并行实现**

```python
class ParallelRecaller:
    """并行多路召回器"""

    def __init__(self):
        self.vector_retriever = VectorRetriever()
        self.bm25_retriever = BM25Retriever()
        self.graph_retriever = GraphRetriever()

    async def recall(
        self,
        query: str,
        top_k_per_path: int = 20
    ) -> Dict[str, List[RetrievalResult]]:
        """三路并行召回"""

        # 创建三个召回任务
        vector_task = asyncio.create_task(
            self.vector_retriever.search(query, top_k_per_path)
        )
        bm25_task = asyncio.create_task(
            self.bm25_retriever.search(query, top_k_per_path)
        )
        graph_task = asyncio.create_task(
            self.graph_retriever.search(query, top_k_per_path)
        )

        # 并行执行并等待所有结果
        try:
            # 设置总体超时(最长等待10秒)
            results = await asyncio.wait_for(
                asyncio.gather(
                    vector_task, bm25_task, graph_task,
                    return_exceptions=True  # 异常不中断其他任务
                ),
                timeout=10.0
            )

            # 处理结果和异常
            output = {}
            for i, (name, result) in enumerate(zip(
                ["vector", "bm25", "graph"], results
            )):
                if isinstance(result, Exception):
                    logger.error(f"{name} recall failed: {result}")
                    output[name] = []
                else:
                    output[name] = result

            return output

        except asyncio.TimeoutError:
            # 超时后取消未完成的任务
            for task in [vector_task, bm25_task, graph_task]:
                if not task.done():
                    task.cancel()
            logger.warning("Multi-path recall timeout")
            return {"vector": [], "bm25": [], "graph": []}

# 使用示例
recaller = ParallelRecaller()
results = await recaller.recall("人工智能发展趋势", top_k_per_path=30)

print(f"Vector: {len(results['vector'])} results")
print(f"BM25: {len(results['bm25'])} results")
print(f"Graph: {len(results['graph'])} results")
```

```python
class WeightedParallelRecaller:
    """动态权重并行召回"""

    async def recall_with_weights(
        self,
        query: str,
        weights: Dict[str, float] = None,
        top_k: int = 30
    ):
        """根据权重分配召回数量"""

        # 默认权重
        if weights is None:
            weights = {"vector": 0.5, "bm25": 0.3, "graph": 0.2}

        # 根据权重分配各路召回数量
        allocated = {
            path: int(top_k * weight * 1.5)  # 多召回50%用于融合
            for path, weight in weights.items()
        }

        # 并行召回
        results = await self._parallel_recall(query, allocated)

        # 根据权重进行结果融合
        fused = self._weighted_fusion(results, weights, top_k)

        return fused

    def _weighted_fusion(
        self,
        results: Dict[str, List],
        weights: Dict[str, float],
        top_k: int
    ) -> List[RetrievalResult]:
        """加权融合"""
        scored_results = []

        for path, path_results in results.items():
            weight = weights.get(path, 0.33)

            for result in path_results:
                # 加权分数 = 原始分数 × 权重
                weighted_score = result.score * weight
                scored_results.append(
                    RetrievalResult(
                        id=result.id,
                        content=result.content,
                        score=weighted_score,
                        metadata={**result.metadata, "source": path}
                    )
                )

        # 按加权分数排序
        scored_results.sort(key=lambda x: x.score, reverse=True)

        return scored_results[:top_k]
```

```python
class RecallQualityChecker:
    """召回质量检查"""

    def check_recall_balance(
        self,
        results: Dict[str, List],
        min_ratio: float = 0.3
    ) -> Dict[str, Any]:
        """检查召回结果均衡性"""

        counts = {path: len(items) for path, items in results.items()}
        max_count = max(counts.values())
        min_count = min(counts.values())

        report = {
            "counts": counts,
            "is_balanced": True,
            "warnings": []
        }

        # 检查各路召回比例
        for path, count in counts.items():
            if count < max_count * min_ratio:
                report["is_balanced"] = False
                report["warnings"].append(
                    f"{path} recall count ({count}) is too low "
                    f"compared to max ({max_count})"
                )

        # 检查召回空结果
        for path, count in counts.items():
            if count == 0:
                report["warnings"].append(f"{path} returned 0 results")

        return report
```

```python
# 内存监控与流式召回
class StreamingRecaller:
    """流式召回,避免内存溢出"""

    def __init__(self, max_memory_mb: int = 500):
        self.max_memory = max_memory_mb * 1024 * 1024  # 转换为字节

    async def recall_streaming(
        self,
        query: str,
        batch_size: int = 10
    ):
        """流式召回,分批返回"""

        results_queue = asyncio.Queue(maxsize=100)

        async def producer():
            """生产者: 各路召回"""
            tasks = [
                self._recall_and_enqueue(
                    path, query, results_queue, batch_size
                )
                for path in ["vector", "bm25", "graph"]
            ]
            await asyncio.gather(*tasks)

            # 生产完成,发送结束信号
            await results_queue.put(None)

        async def consumer():
            """消费者: 处理结果"""
            batch = []
            while True:
                item = await results_queue.get()

                if item is None:  # 结束信号
                    if batch:
                        yield batch
                    break

                batch.append(item)

                # 达到批量大小时返回
                if len(batch) >= batch_size:
                    yield batch
                    batch = []

                # 内存检查
                import sys
                current_memory = sys.getsizeof(batch)
                if current_memory > self.max_memory:
                    logger.warning(
                        f"Memory limit reached: {current_memory} bytes"
                    )
                    yield batch
                    batch = []

        # 启动生产者和消费者
        producer_task = asyncio.create_task(producer())

        async for batch in consumer():
            yield batch

        await producer_task
```

#### 题目14: RRF融合的参数调优策略 ⭐⭐

## 参考答案

### 一、原理阐述

**RRF(Reciprocal Rank Fusion)算法**:

RRF是一种简单高效的融合排序算法,基于排名而非分数进行融合,避免不同召回路径分数分布不一致的问题。

**数学公式**:

```
RRF_score(d) = Σ (1 / (k + rank_i(d)))
```

其中:
- d: 文档
- rank_i(d): 文档d在第i个召回路径中的排名(从1开始)
- k: 平滑参数(通常取60)
- i: 第i个召回路径

**参数k的作用**:
- k值越小,高排名文档权重越大(偏向置信度高的召回)
- k值越大,排名影响被平滑(避免单路召回主导)
- 经验值k=60在多数场景表现良好

**RRF优势**:
1. **无需分数归一化**: 仅依赖排名,避免分数分布差异
2. **鲁棒性强**: 对单路召回异常不敏感
3. **计算高效**: 无需复杂模型,实时性好

### 二、场景应用

**案例1:自适应RRF参数调优**

```python
class AdaptiveRRF:
    """自适应RRF融合"""

    def __init__(self):
        self.default_k = 60
        self.k_by_scenario = {
            "high_precision": 30,  # 偏向高置信度
            "balanced": 60,        # 平衡模式
            "high_recall": 100     # 偏向召回率
        }

    def fuse(
        self,
        multi_path_results: Dict[str, List[RetrievalResult]],
        scenario: str = "balanced",
        top_k: int = 20
    ) -> List[RetrievalResult]:
        """自适应RRF融合"""

        # 根据场景选择k值
        k = self.k_by_scenario.get(scenario, self.default_k)

        # 计算RRF分数
        doc_scores = {}
        doc_contents = {}

        for path, results in multi_path_results.items():
            for rank, result in enumerate(results, start=1):
                doc_id = result.id

                # 累加RRF分数
                rrf_score = 1.0 / (k + rank)
                if doc_id not in doc_scores:
                    doc_scores[doc_id] = 0.0
                    doc_contents[doc_id] = result

                doc_scores[doc_id] += rrf_score

        # 按RRF分数排序
        sorted_docs = sorted(
            doc_scores.items(),
            key=lambda x: x[1],
            reverse=True
        )

        # 返回Top-K结果
        fused_results = []
        for doc_id, score in sorted_docs[:top_k]:
            result = doc_contents[doc_id]
            fused_results.append(
                RetrievalResult(
                    id=doc_id,
                    content=result.content,
                    score=score,
                    metadata={
                        **result.metadata,
                        "rrf_score": score,
                        "fusion_method": f"RRF(k={k})"
                    }
                )
            )

        return fused_results

    def auto_tune_k(
        self,
        query: str,
        ground_truth: List[str],
        k_candidates: List[int] = [30, 60, 100]
    ) -> int:
        """自动调优k参数"""

        best_k = self.default_k
        best_score = 0.0

        for k in k_candidates:
            # 使用当前k值融合
            fused = self.fuse(
                self.recall(query),
                k=k
            )

            # 计算召回率
            retrieved_ids = [r.id for r in fused]
            hit_count = len(set(retrieved_ids) & set(ground_truth))
            recall = hit_count / len(ground_truth) if ground_truth else 0

            if recall > best_score:
                best_score = recall
                best_k = k

        logger.info(
            f"Auto-tuned k={best_k} with recall={best_score:.2%}"
        )
        return best_k
```

```python
class WeightedRRF:
    """带权重的RRF融合"""

    def fuse_with_weights(
        self,
        multi_path_results: Dict[str, List[RetrievalResult]],
        path_weights: Dict[str, float] = None,
        k: int = 60,
        top_k: int = 20
    ):
        """加权RRF融合"""

        # 默认权重
        if path_weights is None:
            path_weights = {"vector": 0.5, "bm25": 0.3, "graph": 0.2}

        # 归一化权重
        total_weight = sum(path_weights.values())
        path_weights = {
            path: w / total_weight for path, w in path_weights.items()
        }

        # 计算加权RRF分数
        doc_scores = {}
        doc_contents = {}

        for path, results in multi_path_results.items():
            weight = path_weights.get(path, 1.0)

            for rank, result in enumerate(results, start=1):
                doc_id = result.id

                # 加权RRF分数
                weighted_rrf = weight / (k + rank)

                if doc_id not in doc_scores:
                    doc_scores[doc_id] = 0.0
                    doc_contents[doc_id] = result

                doc_scores[doc_id] += weighted_rrf

        # 排序并返回
        sorted_docs = sorted(
            doc_scores.items(),
            key=lambda x: x[1],
            reverse=True
        )

        return self._build_results(sorted_docs[:top_k], doc_contents)
```

```python
class QualityAwareRRF:
    """质量感知RRF融合"""

    def fuse_with_quality_filter(
        self,
        multi_path_results: Dict[str, List],
        min_relevance_score: float = 0.5,
        k: int = 60
    ):

        for path, results in multi_path_results.items():

                if r.score >= min_relevance_score

            logger.info(

                f"low quality results"

        for path, results in filtered_results.items():
            if len(results) < 5:
                logger.warning(
                    f"{path} has only {len(results)} results "
                    f"after filtering, consider lowering threshold"
                )

        # 步骤3: 执行RRF融合
        return self.rrf_fuse(filtered_results, k=k)

    def dynamic_k_adjustment(
        self,
        multi_path_results: Dict[str, List]
    ) -> int:
        """根据召回质量动态调整k值"""

        # 计算各路召回的平均分数
        avg_scores = {}
        for path, results in multi_path_results.items():
            if results:
                avg_scores[path] = sum(r.score for r in results) / len(results)

        # 根据平均分数调整k
        overall_avg = sum(avg_scores.values()) / len(avg_scores) if avg_scores else 0.5

        # 高质量召回: 使用小k(偏向高排名)
        # 低质量召回: 使用大k(平滑排名影响)
        if overall_avg > 0.8:
            return 30  # 高置信度场景
        elif overall_avg > 0.5:
            return 60  # 中等置信度
        else:
            return 100  # 低置信度,需要更多探索
```

```python
# 性能优化 - 批量RRF计算
class OptimizedRRF:
    """优化的RRF实现"""

    def fuse_batch(
        self,
        multi_path_results: Dict[str, List],
        k: int = 60,
        top_k: int = 20
    ):
        """优化的批量RRF融合"""

        # 使用numpy加速计算
        import numpy as np
        from collections import defaultdict

        # 预分配分数数组
        doc_scores = defaultdict(float)
        doc_contents = {}

        # 向量化计算
        for path, results in multi_path_results.items():
            # 批量计算RRF分数
            ranks = np.arange(1, len(results) + 1)
            rrf_scores = 1.0 / (k + ranks)

            # 批量更新
            for i, result in enumerate(results):
                doc_id = result.id
                doc_scores[doc_id] += rrf_scores[i]
                if doc_id not in doc_contents:
                    doc_contents[doc_id] = result

        # 使用堆排序而非全排序(性能优化)
        import heapq
        top_docs = heapq.nlargest(
            top_k,
            doc_scores.items(),
            key=lambda x: x[1]
        )

        return [self._build_result(doc_id, score, doc_contents)
                for doc_id, score in top_docs]
```

#### 题目15: HyDE生成策略的优化方案 ⭐⭐

## 参考答案

### 一、原理阐述

**HyDE(Hypothetical Document Embeddings)核心思想**:

HyDE通过LLM生成"假设性答案文档",用这个假设文档的向量去检索真实文档,从而弥合短查询与长文档之间的语义鸿沟。

**工作流程**:
```
用户查询 → LLM生成假设文档 → 假设文档向量化 → 向量检索 → 返回真实文档
```

**核心优势**:
1. **语义对齐**:假设文档与真实文档在语义空间更接近
2. **上下文丰富**:假设文档包含更多相关术语和概念
3. **提升召回**:特别是对于短查询或抽象问题

**优化方向**:
- **生成质量**:控制假设文档的长度、相关性和信息密度
- **计算效率**:减少LLM调用延迟
- **场景适应**:针对不同问题类型调整生成策略

### 二、场景应用

```python
class HyDEQualityEvaluator:
    """HyDE质量评估器"""

    def evaluate_hyde_quality(
        self,
        hypothetical_doc: str,
        retrieved_docs: List[RetrievalResult],
        original_query: str
    ) -> Dict[str, float]:
        """评估HyDE生成质量"""

        metrics = {}

        # 1. 长度合适度(100-300字最佳)
        doc_length = len(hypothetical_doc)
        if 100 <= doc_length <= 300:
            metrics["length_score"] = 1.0
        elif doc_length < 100:
            metrics["length_score"] = doc_length / 100
        else:
            metrics["length_score"] = max(0, 1 - (doc_length - 300) / 300)

        # 2. 术语密度(包含专业术语的比例)
        technical_terms = self._extract_technical_terms(hypothetical_doc)
        term_density = len(technical_terms) / max(doc_length / 10, 1)
        metrics["term_density_score"] = min(term_density, 1.0)

        # 3. 召回相关性(检索结果与查询的平均相关性)
        if retrieved_docs:
            avg_relevance = sum(r.score for r in retrieved_docs) / len(retrieved_docs)
            metrics["retrieval_relevance_score"] = avg_relevance
        else:
            metrics["retrieval_relevance_score"] = 0.0

        # 4. 综合得分
        metrics["overall_score"] = (
            metrics["length_score"] * 0.2 +
            metrics["term_density_score"] * 0.3 +
            metrics["retrieval_relevance_score"] * 0.5
        )

        return metrics

    def _extract_technical_terms(self, text: str) -> List[str]:
        """提取技术术语"""
        import re
        # 简单规则:连续大写字母或专业词汇
        terms = re.findall(r'\b[A-Z]{2,}\b|[A-Za-z]+(?:\s+[A-Za-z]+)*', text)
        return [t for t in terms if len(t) > 3]

class QualityAwareHyDEGenerator:
    """质量感知的HyDE生成器"""

    def __init__(self):

    async def generate_with_quality_check(
        self,
        query: str,
        max_retries: int = 2
    ) -> str:
        """生成并检查质量"""

        for attempt in range(max_retries):
            # 生成假设文档
            hypothetical_doc = await self._generate(query)

            # 质量评估
            metrics = self.evaluator.evaluate_hyde_quality(
                hypothetical_doc,
                [],  # 初始生成时无检索结果
                query
            )

            # 质量达标则返回
            if metrics["overall_score"] >= 0.6:
                self.quality_history.append({
                    "query": query,
                    "doc": hypothetical_doc,
                    "metrics": metrics
                })
                return hypothetical_doc

            # 质量不达标,调整策略重试
            if attempt < max_retries - 1:
                logger.info(
                    f"HyDE quality {metrics['overall_score']:.2f} "
                    f"< 0.6, retrying..."
                )

                await self._adjust_generation_params(metrics)

        # 重试失败,返回最后生成的文档
        return hypothetical_doc
```

```python
        analysis = {}

        analysis["hyde_doc_terms"] = len(self._extract_terms(hyde_doc))

        # 检查检索结果差异
        baseline_ids = {r.id for r in baseline_results}
        hyde_ids = {r.id for r in hyde_results}
        analysis["overlap_ratio"] = len(baseline_ids & hyde_ids) / max(
            len(baseline_ids), 1
        )

        # 判断失败类型
        if analysis["hyde_doc_length"] < 50:
            analysis["failure_type"] = "too_short"
            analysis["suggestion"] = "增加假设文档长度"
        elif analysis["overlap_ratio"] < 0.3:
            analysis["failure_type"] = "semantic_drift"
            analysis["suggestion"] = "假设文档偏离查询语义"
        else:
            analysis["failure_type"] = "unknown"
            analysis["suggestion"] = "考虑禁用HyDE或调整生成策略"

        return analysis

        self,
        retrieved_ids: List[str],
        ground_truth_ids: List[str]
    ) -> float:
        """计算召回率"""
        if not ground_truth_ids:
            return 0.0
        hits = len(set(retrieved_ids) & set(ground_truth_ids))
        return hits / len(ground_truth_ids)
```

#### 题目16: 向量检索与图谱检索的结果冲突解决 ⭐⭐

## 参考答案

### 一、原理阐述

**冲突场景**:

当向量检索和图谱检索返回的结果不一致或矛盾时,需要融合策略解决冲突。

**典型冲突类型**:
1. **实体冲突**:向量检索返回实体A,图谱推理指向实体B
2. **关系冲突**:向量结果暗示关系X,图谱路径显示关系Y
3. **置信度冲突**:向量分数高但图谱证据弱,或反之

**冲突解决策略**:
- **加权融合**:根据问题类型和置信度动态调整权重
- **证据叠加**:图谱关系作为证据增强向量结果

```python
class EntityConflictResolver:
    """实体冲突解决器"""

    def resolve_entity_conflict(
        self,
        vector_entities: List[ExtractedEntity],
        graph_entities: List[ExtractedEntity],
        query: str
    ) -> List[ExtractedEntity]:
        """解决实体冲突"""

        # 1. 实体对齐(同名不同ID)
        aligned = self._align_entities(vector_entities, graph_entities)

        # 2. 置信度融合
        fused_entities = []
        for entity_pair in aligned:
            vector_ent = entity_pair.get("vector")
            graph_ent = entity_pair.get("graph")

            if vector_ent and graph_ent:
                # 都存在,取加权平均
                fused_confidence = (
                    vector_ent.confidence * 0.6 +
                    graph_ent.confidence * 0.4
                )
                fused_entities.append(ExtractedEntity(
                    id=vector_ent.id,  # 使用向量ID
                    name=vector_ent.name,
                    type=vector_ent.type,
                    confidence=fused_confidence,
                    source="fused"
                ))
            elif vector_ent:
                # 仅向量有
                fused_entities.append(vector_ent)
            elif graph_ent:
                # 仅图谱有,降低置信度
                graph_ent.confidence *= 0.8  # 惩罚孤立证据
                fused_entities.append(graph_ent)

        return fused_entities

    def _align_entities(
        self,
        vector_entities: List[ExtractedEntity],
        graph_entities: List[ExtractedEntity]
    ) -> List[Dict]:
        """实体对齐"""

        aligned = []
        matched_graph_indices = set()

        # 按名称匹配
        for v_ent in vector_entities:
            match_found = False
            for i, g_ent in enumerate(graph_entities):
                if i in matched_graph_indices:
                    continue

                if self._is_same_entity(v_ent, g_ent):
                    aligned.append({"vector": v_ent, "graph": g_ent})
                    matched_graph_indices.add(i)
                    match_found = True
                    break

            if not match_found:
                aligned.append({"vector": v_ent, "graph": None})

        # 剩余未匹配的图谱实体
        for i, g_ent in enumerate(graph_entities):
            if i not in matched_graph_indices:
                aligned.append({"vector": None, "graph": g_ent})

        return aligned

    def _is_same_entity(self, e1: ExtractedEntity, e2: ExtractedEntity) -> bool:
        """判断是否同一实体"""
        # 名称完全匹配
        if e1.name == e2.name:
            return True

        # 名称相似度
        from difflib import SequenceMatcher
        similarity = SequenceMatcher(None, e1.name, e2.name).ratio()
        if similarity > 0.8:
            return True

        # 类型匹配 + 名称部分匹配
        if e1.type == e2.type and similarity > 0.6:
            return True

        return False
```

```python
class RelationConflictResolver:

    def resolve_relation_conflict(
        self,
        query: str,
        vector_hints: List[str],
        graph_paths: List[GraphPath]
    ) -> Dict[str, Any]:
        """解决关系冲突"""

        # 1. 从向量结果提取关系提示
        vector_relations = self._extract_relation_hints(vector_hints)

        # 2. 从图谱路径提取关系
        graph_relations = [
            {"path": p, "relations": self._extract_path_relations(p)}
            for p in graph_paths
        ]

        # 3. 关系匹配与验证
        validation_results = []
        for v_rel in vector_relations:
            is_supported = False
            supporting_paths = []

            for g_rel in graph_relations:
                if self._relation_matches(v_rel, g_rel["relations"]):
                    is_supported = True
                    supporting_paths.append(g_rel["path"])

            validation_results.append({
                "relation": v_rel,
                "is_supported": is_supported,
                "supporting_paths": supporting_paths,
                "confidence": 1.0 if is_supported else 0.3
            })

        # 4. 生成最终关系集合
        final_relations = []
        for result in validation_results:
            if result["is_supported"]:
                # 图谱支持,高置信度
                final_relations.append({
                    "relation": result["relation"],
                    "confidence": 0.9,
                    "evidence": "graph_supported"
                })
            else:
                # 仅向量提示,低置信度
                final_relations.append({
                    "relation": result["relation"],
                    "confidence": 0.4,
                    "evidence": "vector_only"
                })

        return {
            "relations": final_relations,
            "conflicts_resolved": len([
                r for r in validation_results if not r["is_supported"]
            ])
        }

    def _extract_relation_hints(self, vector_hints: List[str]) -> List[str]:
        """从向量结果提取关系提示"""
        import re

        relations = []
        relation_patterns = [
            r'(.+?)负责(.+)',
            r'(.+?)管理(.+)',
            r'(.+?)属于(.+)',
            r'(.+?)位于(.+)',
        ]

        for hint in vector_hints:
            for pattern in relation_patterns:
                matches = re.findall(pattern, hint)
                if matches:
                    relations.extend([f"{m[0]}->{m[1]}" for m in matches])

        return relations

    def _relation_matches(self, vector_rel: str, graph_rels: List[str]) -> bool:
        """判断关系是否匹配"""
        # 简化匹配逻辑
        return any(vector_rel in g_rel for g_rel in graph_rels)
```

```python
class IntelligentConflictResolver:

    def __init__(self):

            strategy = best_match["strategy"]

            logger.info(

        else:

            "resolution": resolution,
            "context": context,
            "timestamp": datetime.now()
        })

        return resolution

        conflict: Dict,

        请确认是否正确 (yes/no):

        # 通过WebSocket实时询问用户
        user_response = await self._ask_user_via_websocket(prompt)

        resolution = initial_resolution.copy()
        if user_response == "yes":
            resolution["confidence"] = 1.0
            resolution["source"] = "user_confirmed"
        else:

            resolution["source"] = "user_rejected"

                "rejected_resolution": initial_resolution,
                "timestamp": datetime.now()

                    "score": score
                })
```

## 参考答案

### 一、原理阐述

**监控埋点体系**:

在七层架构的每个层级设置性能监控点,收集关键指标,为性能优化和问题排查提供数据支持。

**监控指标分类**:
1. **延迟指标**:各层处理耗时
2. **吞吐指标**:QPS、并发数
3. **错误率**:失败请求比例
4. **资源指标**:内存、CPU、连接数

**七层埋点位置**:

| 层级 | 关键指标 | 埋点位置 |
|-----|---------|---------|
| L1-API网关层 | 请求延迟、限流次数 | middleware |
| L2-认证授权层 | Token验证耗时 | auth_middleware |
| L3-路由分发层 | 路由匹配耗时 | router |
| L4-业务服务层 | 业务逻辑耗时 | service methods |
| L5-数据访问层 | 数据库查询耗时 | repository |
| L6-基础设施层 | 缓存命中率、连接数 | connector |
| L7-外部服务层 | LLM调用耗时 | llm_client |

```python
@dataclass
class MetricData:

    operation: str
    duration_ms: float

class PerformanceMonitor:
    """性能监控器"""

    def __init__(self):
        self.metrics = []
        self.prometheus_exporter = None  # Prometheus导出器

    def track(
        self,
        layer: str,
        operation: str
    ) -> Callable:
        """性能追踪装饰器"""

        def decorator(func: Callable) -> Callable:
            @wraps(func)
            async def async_wrapper(*args, **kwargs):
                start_time = time.time()
                success = True
                error_type = None

                try:
                    result = await func(*args, **kwargs)
                    return result

                except Exception as e:
                    success = False
                    error_type = type(e).__name__
                    raise

                finally:
                    duration_ms = (time.time() - start_time) * 1000

                    # 记录指标
                    metric = MetricData(
                        layer=layer,
                        operation=operation,
                        duration_ms=duration_ms,
                        success=success,
                        error_type=error_type
                    )
                    self.metrics.append(metric)

                    # 导出到Prometheus
                    if self.prometheus_exporter:
                        self.prometheus_exporter.record(metric)

            return async_wrapper

        return decorator
```

```python
class PerformanceDashboard:
    """性能监控仪表盘"""

    def __init__(self, monitor: PerformanceMonitor):
        self.monitor = monitor
        self.window_size = 1000  # 统计最近1000次请求

    def get_layer_stats(self, layer: str) -> Dict[str, Any]:
        """获取指定层级的统计信息"""

        # 过滤最近指标
        recent_metrics = [
            m for m in self.monitor.metrics[-self.window_size:]
            if m.layer == layer
        ]

        if not recent_metrics:
            return {"error": "No data available"}

        # 计算统计信息
        durations = [m.duration_ms for m in recent_metrics]
        success_rate = sum(1 for m in recent_metrics if m.success) / len(recent_metrics)

        # 错误类型分布
        error_distribution = {}
        for m in recent_metrics:
            if m.error_type:
                error_distribution[m.error_type] = \
                    error_distribution.get(m.error_type, 0) + 1

        return {
            "layer": layer,
            "avg_latency_ms": sum(durations) / len(durations),
            "p50_latency_ms": self._percentile(durations, 0.5),
            "p95_latency_ms": self._percentile(durations, 0.95),
            "p99_latency_ms": self._percentile(durations, 0.99),
            "throughput_qps": len(recent_metrics) / self.window_size,
            "success_rate": success_rate,
            "error_distribution": error_distribution,
            "operation_breakdown": self._group_by_operation(recent_metrics)
        }

    def generate_report(self) -> str:
        """生成性能报告"""

        report_lines = ["=" * 60]
        report_lines.append("KQuery Pro Performance Report")
        report_lines.append("=" * 60)
        report_lines.append(f"Time: {datetime.now()}")
        report_lines.append("")

        for layer in ["L1", "L2", "L3", "L4", "L5", "L6", "L7"]:
            stats = self.get_layer_stats(layer)

            if "error" in stats:
                report_lines.append(f"{layer}: No data")
                continue

            report_lines.append(f"{layer} Layer:")
            report_lines.append(f"  Avg Latency: {stats['avg_latency_ms']:.2f}ms")
            report_lines.append(f"  P95 Latency: {stats['p95_latency_ms']:.2f}ms")
            report_lines.append(f"  Success Rate: {stats['success_rate']:.1%}")
            report_lines.append("")

        return "\n".join(report_lines)

    def _percentile(self, data: List[float], p: float) -> float:
        """计算百分位数"""
        sorted_data = sorted(data)
        k = (len(sorted_data) - 1) * p
        f = int(k)
        c = f + 1 if f + 1 < len(sorted_data) else f
        return sorted_data[int(k)]

    def _group_by_operation(self, metrics: List[MetricData]) -> Dict:
        """按操作分组统计"""
        groups = {}
        for m in metrics:
            if m.operation not in groups:
                groups[m.operation] = []
            groups[m.operation].append(m.duration_ms)

        return {
            op: {
                "avg_ms": sum(durs) / len(durs),
                "count": len(durs)
            }
            for op, durs in groups.items()
        }
```

```python
# 可靠的监控埋点
class ReliablePerformanceMonitor:
    """可靠的性能监控器"""

    def __init__(self):
        self.local_buffer = []  # 本地缓冲区
        self.max_buffer_size = 10000

    def track_reliable(self, layer: str, operation: str):
        """可靠的埋点装饰器"""

        def decorator(func):
            @wraps(func)
            async def wrapper(*args, **kwargs):
                metric_id = str(uuid.uuid4())
                start_time = time.time()
                success = True
                error_type = None

                try:
                    result = await func(*args, **kwargs)
                    return result

                except Exception as e:
                    success = False
                    error_type = type(e).__name__
                    raise

                finally:
                    # 无论成功失败都记录
                    duration_ms = (time.time() - start_time) * 1000

                    metric = {
                        "id": metric_id,
                        "layer": layer,
                        "operation": operation,
                        "duration_ms": duration_ms,
                        "success": success,
                        "error_type": error_type,
                        "timestamp": datetime.now().isoformat()
                    }

                    # 优先写入本地缓冲
                    self.local_buffer.append(metric)

                    # 缓冲区溢出检查
                    if len(self.local_buffer) > self.max_buffer_size:
                        logger.warning(
                            f"Metric buffer overflow, "
                            f"dropping {len(self.local_buffer) - self.max_buffer_size} metrics"
                        )
                        self.local_buffer = self.local_buffer[-self.max_buffer_size:]

                    # 异步上报(失败不影响主流程)
                    try:
                        asyncio.create_task(self._report_async(metric))
                    except Exception as e:
                        logger.error(f"Metric report failed: {e}")

            return wrapper

        return decorator

    async def _report_async(self, metric: Dict):
        """异步上报指标"""

        # 尝试上报到Prometheus
        try:
            if self.prometheus_exporter:
                self.prometheus_exporter.record(metric)
        except Exception as e:
            logger.warning(f"Prometheus report failed: {e}")

        # 同时写入本地日志作为备份
        try:
            with open("metrics_backup.log", "a") as f:
                f.write(json.dumps(metric) + "\n")
        except Exception as e:
            logger.error(f"Backup metric log failed: {e}")
```
