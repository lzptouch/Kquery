```typescript
import React, { useState, ChangeEvent, FormEvent } from 'react';
import { Button } from './ui/button';
import { Input } from './ui/input';

interface DocumentUploadProps {
  onUploadSuccess?: (files: File[]) => void;
  onError?: (error: Error) => void;
  maxSize?: number;
}

export const DocumentUpload: React.FC<DocumentUploadProps> = ({
  onUploadSuccess,
  onError,
  maxSize = 50 * 1024 * 1024,
}) => {
  const [files, setFiles] = useState<File[]>([]);
  const [uploading, setUploading] = useState(false);
  
  const handleFileChange = (event: ChangeEvent<HTMLInputElement>) => {
    const selectedFiles = event.target.files;
    if (selectedFiles) {
      setFiles(Array.from(selectedFiles));
    }
  };
  
  const handleUpload = async () => {
    try {
      setUploading(true);
      // 上传逻辑
      onUploadSuccess?.(files);
    } catch (error) {
      onError?.(error as Error);
    } finally {
      setUploading(false);
    }
  };
  
  return (
    <div className="document-upload">
      <input
        type="file"
        multiple
        onChange={handleFileChange}
        disabled={uploading}
      />
      <Button
        onClick={handleUpload}
        disabled={uploading || files.length === 0}
      >
        {uploading ? '上传中...' : '上传文档'}
      </Button>
    </div>
  );
};
```

#### 2.3.2 自定义Hooks

```typescript
import { useState, useEffect } from 'react';

interface UseQueryResult {
  query: string;
  results: RetrievalResult[];
  isLoading: boolean;
  error: Error | null;
  executeQuery: (query: string) => Promise<void>;
}

export const useQuery = (collectionName: string): UseQueryResult => {
  const [query, setQuery] = useState<string>('');
  const [results, setResults] = useState<RetrievalResult[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  
  const executeQuery = async (queryText: string) => {
    try {
      setIsLoading(true);
      setError(null);
      setQuery(queryText);
      
      const response = await fetch(`/api/v1/kquery/documents/query`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          query: queryText,
          collection_name: collectionName,
          top_k: 5
        })
      });
      
      const data = await response.json();
      setResults(data.results);
    } catch (err) {
      setError(err as Error);
    } finally {
      setIsLoading(false);
    }
  };
  
  return {
    query,
    results,
    isLoading,
    error,
    executeQuery
  };
};
```

## 5. 异步编程规范

### 5.1 基本原则

项目全面采用 `async/await`，遵循以下原则：

```python
# ✅ 好的做法：使用 async with 管理异步上下文
async def read_file(path: str) -> str:
    async with aiofiles.open(path, 'r') as f:
        return await f.read()

# ✅ 好的做法：使用 asyncio.gather 并行执行
async def fetch_all(ids: List[str]) -> List[Result]:
    tasks = [fetch_one(id) for id in ids]
    return await asyncio.gather(*tasks)

# ✅ 好的做法：使用信号量控制并发
_semaphore = asyncio.Semaphore(10)

async def rate_limited_call():
    async with _semaphore:
        return await external_api_call()

# ❌ 坏的做法：在 async 函数中使用同步阻塞调用
async def bad_example():
    time.sleep(5)  # 使用 await asyncio.sleep(5) 代替
    requests.get(url)  # 使用 aiohttp 代替

# ❌ 坏的做法：无并发控制的并行任务
async def uncontrolled():
    tasks = [call() for _ in range(1000)]  # 可能耗尽资源
    return await asyncio.gather(*tasks)
```

```python
    """使用 lru_cache 确保全局唯一实例"""
    return Settings()

def get_db_session() -> AsyncSession:
    """每次请求创建新的数据库会话"""
    return AsyncSessionLocal()

# ✅ 好的做法：使用生成器管理资源生命周期
async def get_db() -> AsyncGenerator[AsyncSession, None]:
    async with AsyncSessionLocal() as session:
        try:
            yield session
        finally:
            await session.close()

# ❌ 坏的做法：在模块级别创建可变全局状态
global_service = RAGService()  # 禁止：影响测试和并发
```

## 9. Git工作流规范

### 9.1 分支策略

```bash
# 主分支：main
main          # 生产环境，稳定版本

# 开发分支：develop
develop       # 开发环境，最新功能

# 功能分支：feature/*
feature/kg   # 知识图谱功能
feature/auth # 认证授权功能

# 修复分支：fix/*
fix/cache   # 缓存bug修复
fix/auth    # 认证bug修复

# 发布分支：release/*
release/v1.0.0  # 1.0版本发布
```

### 9.2 Commit消息规范

使用 **Conventional Commits** 格式：

```bash
# 格式：<type>(<scope>): <subject>

# type: feat, fix, docs, style, refactor, test, chore, perf, security
# scope: 模块/功能
# subject: 简短描述
```

**类型说明**：

| 类型 | 用途 |
|------|------|
| `feat` | 新功能 |
| `fix` | 修复bug |
| `docs` | 文档更新 |
| `style` | 代码格式调整（不影响功能） |
| `refactor` | 重构（不改变功能） |
| `test` | 测试相关 |
| `chore` | 构建/工具链/依赖升级 |
| `perf` | 性能优化 |
| `security` | 安全修复 |

**示例**：

```bash
feat(auth): implement JWT authentication system
fix(graph): resolve node duplication issue in graph builder
docs(api): update swagger documentation for graph endpoints
perf(cache): optimize Redis cache hit rate to 70%
security(rbac): fix permission bypass vulnerability
```

**详细说明（可选）**：

```bash
feat(rag): 添加HyDE假设性文档生成功能

- 实现hyde_generator.py
- 集成到rag_service.py
- 添加配置开关RAG_USE_HYDE

Closes #123
```

### 9.3 工作流

